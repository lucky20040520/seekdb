cmake_minimum_required(VERSION 3.16)
project(minivector_cpp LANGUAGES CXX)
find_package(nlohmann_json 3.11 REQUIRED)
add_executable(minivector main.cpp storage.cpp sql.cpp query.cpp)
target_compile_features(minivector PRIVATE cxx_std_17)
target_compile_options(minivector PRIVATE -ffp-contract=off -Wall -Wextra -Wpedantic)
target_link_libraries(minivector PRIVATE nlohmann_json::nlohmann_json)
install(TARGETS minivector RUNTIME DESTINATION bin)
